{{- $cfg := hugo.Data.github -}} {{- $fallback := hugo.Data.github_fallback -}} {{- $username := $cfg.username | default "user27c" -}} {{- $excludeForks := $cfg.excludeForks | default true -}} {{- $excludeRepos := $cfg.excludeRepos | default (slice) -}} {{- $maxRepos := $cfg.maxRepos | default 50 -}} {{- $sortBy := $cfg.sortBy | default "updated" -}} {{- $user := $fallback.user | default dict -}} {{- $repos := slice -}} {{- $fetchError := "" -}} {{- with $fallback.repos -}} {{- range . -}} {{- $skip := false -}} {{- if and $excludeForks .fork }}{{ $skip = true }}{{ end -}} {{- if in $excludeRepos .name }}{{ $skip = true }}{{ end -}} {{- if not $skip }}{{ $repos = $repos | append . }}{{ end -}} {{- end -}} {{- else -}} {{- $fetchError = "GitHub snapshot unavailable" -}} {{- end -}} {{- if eq $sortBy "stars" -}} {{- $repos = sort $repos "stargazers_count" "desc" -}} {{- else -}} {{- $repos = sort $repos "updated_at" "desc" -}} {{- end -}} {{- if gt (len $repos) $maxRepos }}{{ $repos = first $maxRepos $repos }}{{ end -}} {{- .Scratch.Set "github" (dict "user" $user "repos" $repos "username" $username "error" $fetchError) -}}
